pragma solidity ^0.8.0;

// Bank Account Smart Contract
contract BankAccount {

    // Store account balance
    mapping(address => uint) public balance;

    // Deposit money
    function deposit() public payable {
        require(msg.value > 0, "Amount must be greater than zero");

        balance[msg.sender] += msg.value;
    }

    // Withdraw money
    function withdraw(uint amount) public {
        require(amount > 0, "Invalid amount");

        require(balance[msg.sender] >= amount,
                "Insufficient balance");

        balance[msg.sender] -= amount;

        payable(msg.sender).transfer(amount);
    }

    // Show account balance
    function showBalance() public view returns (uint) {
        return balance[msg.sender];
    }
}

pragma solidity ^0.8.0;

contract Bank {

    // Store customer balances
    mapping(address => uint) public balanceOf;

    // Deposit money
    function deposit() public payable {
        require(msg.value > 0, "Deposit must be greater than zero");

        balanceOf[msg.sender] += msg.value;
    }

    // Withdraw entire balance
    function withdraw() public payable {

        // Do not accept additional ETH during withdrawal
        require(msg.value == 0, "Do not send ETH while withdrawing");

        uint amount = balanceOf[msg.sender];

        require(amount > 0, "Insufficient balance");

        // Update balance before transferring ETH
        balanceOf[msg.sender] = 0;

        // Transfer money to customer
        payable(msg.sender).transfer(amount);
    }

    // Show account balance
    function showBalance() public view returns (uint) {
        return balanceOf[msg.sender];
    }
}

Open Remix IDE.
On the left side, click File Explorer.

Click the New File icon.

Paste your Bank Account Smart Contract code.

BankAccount

3. Compile the Smart Contract
Click the Solidity Compiler icon on the left.
Select compiler version 0.8.x (for example, 0.8.20).
Click Compile BankAccount.sol.
Wait for the green tick.

If compilation succeeds, your smart contract is ready for deployment.

4. Deploy the Smart Contract
Click Deploy & Run Transactions on the left.
Under Environment, select:
Remix VM (Cancun)

BankAccount

Your contract will be deployed.

You should see:

Deployed Contracts

BankAccount at 0x...

Remix VM is a simulated blockchain, so you don't need real ETH or MetaMask for this practical.

8. Important Viva Questions

Q1. Where do we execute Solidity programs?

Using Remix IDE.

Q2. What is Solidity?

Solidity is a high-level programming language used to develop smart contracts on Ethereum-compatible blockchains.

Q3. What is a smart contract?

A smart contract is a program deployed on a blockchain that automatically executes predefined instructions.

Q4. What is Remix VM?

It is a simulated blockchain environment provided by Remix IDE for testing smart contracts.

Q5. What does payable mean?

It allows a function to receive Ether.

Q6. What is msg.sender?

It represents the address of the account that called the function.

Q7. What is msg.value?

It represents the amount of Ether sent with the current transaction, measured in Wei.

Q8. What is mapping in Solidity?

Mapping is a key-value data structure used to store information. Here, it stores the balance associated with each address.

Q9. What is require()?

It checks a condition. If the condition is false, the transaction is reverted.

Q10. What is Wei?

Wei is the smallest unit of Ether.

1 ETH = 10^18 Wei
